\gentry{connected-component}{Connected component}{%
  A group of places in the neighbour graph that can all reach one another by
  hopping from neighbour to neighbour, and which is joined to nothing outside
  the group. A graph with two components has been cut in two.}

\gentry{null-hypothesis}{Null hypothesis}{%
  The deliberately dull explanation a statistical test tries to contradict ---
  here, that the observed values were scattered across the map with no regard
  to location.}

\gentry{p-value}{p-value}{%
  The probability of seeing a result at least as extreme as the one observed,
  computed on the assumption that the null hypothesis is true. A small p-value
  says the observation is unusual under the dull explanation; it is not the
  probability that the dull explanation is wrong.}

\gentry{spatial-weights-matrix}{Spatial weights matrix}{%
  A square table with one row and one column per place, whose entry in row $i$
  and column $j$ records how strongly place $j$ counts as a neighbour of place
  $i$. It is the object that turns a map into a graph.}
